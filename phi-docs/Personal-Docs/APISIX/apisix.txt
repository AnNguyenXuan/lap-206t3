ETCD_VERSION='3.6.7'

tar -xzf etcd-v${ETCD_VERSION}-linux-amd64.tar.gz
sudo mv etcd-v${ETCD_VERSION}-linux-amd64/etcd /usr/local/bin/
sudo mv etcd-v${ETCD_VERSION}-linux-amd64/etcdctl /usr/local/bin/


nano /etc/systemd/system/etcd.service


# 1) tạo thư mục data và user etcd (khuyến nghị)
sudo useradd -r -s /sbin/nologin etcd || true
sudo mkdir -p /var/lib/etcd
sudo chown -R etcd:etcd /var/lib/etcd

# 2) reload systemd & start
sudo systemctl daemon-reload
sudo systemctl enable --now etcd

# 3) kiểm tra trạng thái & logs
sudo systemctl status etcd --no-pager
sudo journalctl -u etcd -f
