[global]
   workgroup = INLANEFREIGHT
   security = ADS
   realm = INLANEFREIGHT.LOCAL

   server role = member server

   # Kerberos
   dedicated keytab file = /etc/krb5.keytab
   kerberos method = secrets and keytab

   # Winbind
   winbind use default domain = yes
   winbind enum users  = yes
   winbind enum groups = yes
   winbind refresh tickets = yes
   template shell = /bin/bash
   template homedir = /home/%U

   # IDMAP
   idmap config * : backend = tdb
   idmap config * : range   = 10000-99999
   idmap config INLANEFREIGHT : backend = rid
   idmap config INLANEFREIGHT : range   = 100000-999999

[cephshare]
   comment     = CephFS Share via VFS
   path        = /
   read only   = no
   browseable  = yes
   guest ok    = no
   kernel share modes = no
   vfs objects = ceph
   ceph:config_file = /etc/ceph/ceph.conf
   ceph:userid      = client.admin
   ceph:secret      = /etc/ceph/ceph.client.admin.keyring
   valid users = +"Domain Users"
